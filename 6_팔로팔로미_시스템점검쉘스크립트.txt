#!/bin/bash

# 자동화 시 PATH 문제 방지
export PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# 1. 전역 변수 초기화 (최상단에 위치해야 합니다)
TOTAL_COUNT=0
GOOD_COUNT=0
BAD_COUNT=0
MANUAL_COUNT=0

# 2. 결과 출력 및 집계 함수
result_print() {
    local input=("$@")
    local code="${input[0]}"
    local check_content="${input[1]}"
    local overall_result="${input[2]}"
    
    # [핵심] 결과 누적 로직 (여기서 1~67까지 카운트를 셉니다)
    ((TOTAL_COUNT++))
    if [[ "$overall_result" == "취약" ]]; then
        ((BAD_COUNT++))
    elif [[ "$overall_result" == "수동점검" ]]; then
        ((MANUAL_COUNT++))
    else
        ((GOOD_COUNT++))
    fi

    # 기존 출력 로직
    local red="\e[31m"
    local green="\e[32m"
    local yellow="\e[33m"
    local reset="\e[0m"

    echo -e "[${yellow}${code}${reset}] ${check_content}"
    echo "----------------------------------------------------------"
    
    local result_color="$green"
    if [[ "$overall_result" == "취약" ]]; then result_color="$red"; fi
    echo -e ": ${result_color}${overall_result}${reset}"

    local total_args=${#input[@]}
    for ((i=3; i<total_args; i+=3)); do

    local detail="${input[$i]}"
    local result="${input[$((i+1))]}"
    local requirement="${input[$((i+2))]}"

    # U_14~U_33처럼 상세내용만 전달된 경우 처리
    if [[ -z "$result" && -z "$requirement" ]]; then
        echo -e "\t [+] ${detail}"
        continue
    fi

    # [수정] 결과가 취약이면 기호를 [!]로, 아니면 [+]로 설정
        local item_symbol="+"
        local item_color="$green"
        if [[ "$result" == "취약" ]]; then
            item_symbol="!"
            item_color="$red"
        fi

        echo -e "\t [${item_color}${item_symbol}${reset}] ${detail}: ${item_color}${result}${reset}"

        if [[ "$overall_result" != "양호" && -n "$requirement" && "$requirement" != "-" ]]; then
            echo -e "\t      [!] 조치사항: ${requirement}"
        fi
    done
    echo ""
}

# [새로 추가할 도구 함수들]
is_enabled_or_active() { systemctl is-enabled "$1" >/dev/null 2>&1 || systemctl is-active "$1" >/dev/null 2>&1; }
is_active() { systemctl is-active "$1" >/dev/null 2>&1; }
listen_tcp() { ss -ntl | grep -q ":$1 "; }
listen_udp() { ss -nul | grep -q ":$1 "; }
owner() { stat -c "%U" "$1" 2>/dev/null; }
mode() { stat -c "%a" "$1" 2>/dev/null; }
mode_le() { [ "$1" -le "$2" ]; }

# --------------------------------------------------------------------
# 스크립트 실행 부분
# --------------------------------------------------------------------
echo -e "\n시스템 점검 테스트를 시작합니다...\n"

# --------------------------------------------------------------------
# [U_01] root 계정 원격 접속 제한
# --------------------------------------------------------------------
U_01()
{
    local output_mode=$1
    local desc="root 계정 원격접속 제한"
    local detail_1="SSH 원격 접속 제한 여부"

    # 우분투 환경 점검 로직 (PermitRootLogin 찾기)
    local root_login=$(grep -i "^PermitRootLogin" /etc/ssh/sshd_config | awk '{print $2}')

    if [[ "$root_login" == "no" ]]; then
        total_result="양호"
        detail_1_result="양호"
        detail_1_order="-"
    else
        total_result="취약"
        detail_1_result="취약"
        detail_1_order="/etc/ssh/sshd_config 파일에서 PermitRootLogin no 설정 필요"
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U_01" "$desc" "$total_result" "$detail_1" "$detail_1_result" "$detail_1_order"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U_01" "$desc" "$total_result" "$detail_1" "$detail_1_result" "$detail_1_order"
    fi
}

# --------------------------------------------------------------------
# [U_02] 비밀번호 관리정책 설정
# --------------------------------------------------------------------
U_02()
{
    local output_mode=$1
    local desc="패스워드 복잡성 설정"
    local total_result="양호"
    
    # 여러 개의 점검 항목을 한 번에 담아서 배열 선언
    local details=()
    local conf_file="/etc/security/pwquality.conf"

    # 1. 패스워드 복잡성 모듈(pwquality) 활성화 여부 확인
    local pam_check=$(grep "pam_pwquality.so" /etc/pam.d/common-password 2>/dev/null | grep -v "^#")
    if [[ -z "$pam_check" ]]; then
        total_result="취약"
        details+=("복잡성 모듈 활성화")
        details+=("취약")
        details+=("/etc/pam.d/common-password 파일에 pam_pwquality.so 설정 필요")
    else
        details+=("복잡성 모듈 활성화")
        details+=("양호")
        details+=("-")
    fi

    # 2. 최소 길이 (minlen) 8자리 이상 확인
    local minlen=$(grep -E "^\s*minlen" "$conf_file" 2>/dev/null | awk -F'=' '{print $2}' | tr -d ' ')
    if [[ -z "$minlen" ]] || [[ "$minlen" -lt 8 ]]; then
        total_result="취약"
        details+=("최소 길이(minlen) 8자리 이상")
        details+=("취약")
        details+=("$conf_file 파일에서 minlen=8 이상 설정 필요")
    else
        details+=("최소 길이(minlen) 8자리 이상")
        details+=("양호")
        details+=("-")
    fi

    # 3. 영문 소문자/대문자/숫자/특수문자 요구 (lcredit, ucredit, dcredit, ocredit) -1 설정 확인
    local credits=("lcredit" "ucredit" "dcredit" "ocredit")
    for cred in "${credits[@]}"; do
        local val=$(grep -E "^\s*$cred" "$conf_file" 2>/dev/null | awk -F'=' '{print $2}' | tr -d ' ')
        if [[ "$val" != "-1" ]]; then
            total_result="취약"
            details+=("$cred (문자종류 조합)")
            details+=("취약")
            details+=("$conf_file 파일에서 $cred=-1 설정 필요")
        else
            details+=("$cred (문자종류 조합)")
            details+=("양호")
            details+=("-")
        fi
    done

    # result_print 함수로 배열 전체를 던져서 한 번에 예쁘게 출력!
    if [[ "$output_mode" == "all" ]]; then
        result_print "U_02" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U_02" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
# [U_03] 계정 잠금 임계값 설정
# --------------------------------------------------------------------
U_03()
{
    local output_mode=$1
    local desc="계정 잠금 임계값 설정"
    local total_result="양호"
    local details=()

    local pam_file="/etc/pam.d/common-auth"
    
    # 우분투(Debian) KISA 기준: pam_tally2.so 또는 pam_faillock.so 에서 deny 값 추출
    local deny_value=$(grep -E "pam_tally.so|pam_tally2.so|pam_faillock.so" "$pam_file" 2>/dev/null | grep -v "^#" | grep -oP 'deny=\K[0-9]+' | head -n 1)

    # 1. 설정이 아예 없는 경우
    if [[ -z "$deny_value" ]]; then
        total_result="취약"
        details+=("계정 잠금 정책(deny) 설정 여부")
        details+=("취약")
        details+=("$pam_file 파일에 pam_faillock.so 및 deny 설정 필요")
        
    # 2. 설정이 있고 10회 이하인 경우 (KISA 양호 기준)
    elif [[ "$deny_value" -le 10 && "$deny_value" -gt 0 ]]; then
        details+=("계정 잠금 임계값 10회 이하 설정")
        details+=("양호")
        details+=("-")
        
    # 3. 설정은 있는데 10회를 초과한 경우 (취약)
    else
        total_result="취약"
        details+=("계정 잠금 임계값 10회 이하 설정")
        details+=("취약")
        details+=("현재 deny=$deny_value 로 설정됨. 10 이하로 변경 필요")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U_03" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U_03" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
# [U_04] 비밀번호 파일 보호
# --------------------------------------------------------------------
U_04()
{
    local output_mode=$1
    local desc="비밀번호 파일 보호 (쉐도우 패스워드 사용 여부)"
    local total_result="양호"
    local details=()

    local passwd_file="/etc/passwd"
    local shadow_file="/etc/shadow"

    # 1. /etc/shadow 파일이 존재하는지 1차 확인
    if [ ! -f "$shadow_file" ]; then
        details+=("쉐도우 패스워드 파일(/etc/shadow) 존재 여부")
        details+=("취약")
        details+=("pwconv 명령어를 실행하여 쉐도우 패스워드를 적용해야 합니다.")
        total_result="취약"
    else
        details+=("쉐도우 패스워드 파일(/etc/shadow) 존재 여부")
        details+=("양호")
        details+=("-")
    fi

    # 2. KISA 기준: /etc/passwd 파일 두 번째 필드가 'x'인지 검사
    # 패스워드 필드($2)가 'x'도 아니고 '*'도 아닌(즉, 평문으로 노출된) 계정 찾기
    local unshadowed_users=$(awk -F: '$2 != "x" && $2 != "*" {print $1}' "$passwd_file")

    if [ -n "$unshadowed_users" ]; then
        details+=("/etc/passwd 암호화 상태 (두 번째 필드 'x' 표시 여부)")
        details+=("취약")
        details+=("암호화되지 않은 계정 존재. 시스템 관리자 권한으로 pwconv 명령어 실행 필요")
        total_result="취약"
    else
        details+=("/etc/passwd 암호화 상태 (두 번째 필드 'x' 표시 여부)")
        details+=("양호")
        details+=("-")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U_04" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U_04" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
# [U_05] root 이외의 UID가 ‘0’ 금지
# --------------------------------------------------------------------
U_05()
{
    local output_mode=$1
    local desc="root 이외의 UID가 '0' 금지"
    local total_result="양호"
    local details=()

    # /etc/passwd 파일에서 UID(3번째 필드)가 0인데, 계정명(1번째 필드)이 'root'가 아닌 계정 추출
    local uid_zero_users=$(awk -F: '$3 == 0 && $1 != "root" {print $1}' /etc/passwd)

    if [[ -z "$uid_zero_users" ]]; then
        details+=("UID가 0인 일반 계정 존재 여부")
        details+=("양호")
        details+=("-")
    else
        total_result="취약"
        details+=("UID가 0인 일반 계정 존재 여부")
        details+=("취약")
        
        # 발견된 취약 계정들의 이름을 가로로 정렬
        local users_formatted=$(echo "$uid_zero_users" | tr '\n' ' ')
        details+=("root 외에 UID가 0인 계정 존재: [ $users_formatted ]. usermod -u 명령으로 UID 변경 필요")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U_05" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U_05" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
# [U_07] 불필요한 계정 제거
# --------------------------------------------------------------------
U_07()
{
    local output_mode=$1
    local desc="불필요한 계정 제거"
    local total_result="양호"
    local details=()

    # KISA가 가이드에서 명시한 '기본 생성되는 불필요한 계정' 리스트
    local unnecessary_accounts=("lp" "uucp" "nuucp" "games")
    local found_accounts=()

    # /etc/passwd 파일에서 해당 계정들이 존재하는지 하나씩 검사
    for account in "${unnecessary_accounts[@]}"; do
        if grep -q "^${account}:" /etc/passwd; then
            found_accounts+=("$account")
        fi
    done

    # 발견된 계정이 0개라면 양호
    if [ ${#found_accounts[@]} -eq 0 ]; then
        details+=("기본 불필요 계정(lp, uucp 등) 존재 여부")
        details+=("양호")
        details+=("-")
    else
        # 하나라도 발견되면 취약 처리
        total_result="취약"
        details+=("기본 불필요 계정(lp, uucp 등) 존재 여부")
        details+=("취약")
        
        # 발견된 계정들을 보기 좋게 쉼표(,)로 연결해서 출력
        local formatted_accounts=$(IFS=, ; echo "${found_accounts[*]}")
        details+=("불필요한 계정 발견: [ $formatted_accounts ]. userdel 명령어로 제거 권고")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-07" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U-07" "$desc" "$total_result" "${details[@]}"
    fi
}

        total_result="취약"
        details+=("root 그룹 내 불필요한 추가 계정 존재 여부")
        details+=("취약")
        details+=("root 그룹에 불필요한 계정 [ $root_members ] 존재. 'gpasswd -d <계정명> root' 명령으로 제거 필요")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-08" "$desc" "$total_result" "${details[@]}"

# --------------------------------------------------------------------
# [U_09] 계정이 존재하지 않는 GID 금지
# --------------------------------------------------------------------
U_09()
{
    local output_mode=$1
    local desc="계정이 존재하지 않는 GID 금지 (불필요한 그룹 제거)"
    local total_result="양호"
    local details=()

    # 1. /etc/passwd 에서 현재 사용 중인 '기본 그룹(Primary GID)' 목록을 쫙 뽑아옵니다.
    local used_gids=$(awk -F: '{print $4}' /etc/passwd | sort -u)
    
    local empty_groups=()

    # 2. /etc/group 에서 GID 1000 이상(일반 그룹)인 녀석들 중, 
    # 멤버가 텅 비어있고, /etc/passwd에서도 안 쓰는 진짜 껍데기 그룹을 찾습니다.
    while IFS=: read -r group_name password gid members; do
        # GID 1000 이상이면서 우분투 기본 그룹(nobody, nogroup)은 예외 처리
        if [[ "$gid" -ge 1000 && "$group_name" != "nogroup" && "$group_name" != "nobody" ]]; then
            # 4번째 필드(멤버)가 비어있는지 1차 확인
            if [[ -z "$members" ]]; then
                # /etc/passwd의 기본 GID로도 안 쓰이는지 2차 확인
                if ! echo "$used_gids" | grep -qx "$gid"; then
                    empty_groups+=("$group_name")
                fi
            fi
        fi
    done < /etc/group

    # 비어있는 그룹이 0개라면 양호!
    if [ ${#empty_groups[@]} -eq 0 ]; then
        details+=("계정이 없는 불필요한 일반 그룹 존재 여부")
        details+=("양호")
        details+=("-")
    else
        # 하나라도 발견되면 취약!
        total_result="취약"
        details+=("계정이 없는 불필요한 일반 그룹 존재 여부")
        details+=("취약")
        
        local formatted_groups=$(IFS=, ; echo "${empty_groups[*]}")
        details+=("불필요한 그룹 발견: [ $formatted_groups ]. 확인 후 groupdel 명령으로 제거 권고")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-09" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U-09" "$desc" "$total_result" "${details[@]}"
    fi
}

U_10()
{
    local output_mode=$1

    local total_result="양호"
    local details=()

    # /etc/passwd 파일에서 3번째 필드(UID)만 뽑아서 정렬 후, 중복된 값만(-d) 추출
    local duplicate_uids=$(awk -F: '{print $3}' /etc/passwd | sort | uniq -d)

    if [[ -z "$duplicate_uids" ]]; then
        details+=("중복된 UID 존재 여부")
        details+=("양호")
        details+=("-")
    else
        total_result="취약"
        details+=("중복된 UID 존재 여부")
        details+=("취약")
        
        # 중복된 UID들을 가로로 예쁘게 정렬
        local uids_formatted=$(echo "$duplicate_uids" | tr '\n' ' ' | sed 's/ $//')
        details+=("중복된 UID 발견: [ $uids_formatted ]. usermod -u 명령으로 고유한 UID 할당 필요")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-10" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U-10" "$desc" "$total_result" "${details[@]}"
    fi
}

# [U_11] 사용자 shell 점검
# --------------------------------------------------------------------
U_11()
{
    local output_mode=$1
    local desc="사용자 shell 점검 (로그인 불필요 계정의 쉘 제한)"
    local total_result="양호"
    local details=()

    # KISA 가이드에 명시된 점검 대상 계정 리스트
    local target_accounts="daemon|bin|sys|adm|listen|nobody|nobody4|noaccess|diag|operator|games|gopher"
    local vulnerable_accounts=()

    # /etc/passwd 에서 해당 계정들을 찾아 쉘(7번째 필드)이 가짜 쉘인지 검사
    while IFS=: read -r user pass uid gid info home shell; do
        if [[ "$shell" != "/bin/false" && "$shell" != "/sbin/nologin" && "$shell" != "/usr/sbin/nologin" ]]; then
            vulnerable_accounts+=("$user")
        fi
    done < <(grep -E "^($target_accounts):" /etc/passwd)

    if [ ${#vulnerable_accounts[@]} -eq 0 ]; then
        details+=("로그인이 불필요한 계정에 /sbin/nologin 쉘 부여 여부")
        details+=("양호")
        details+=("-")
    else
        total_result="취약"
        details+=("로그인이 불필요한 계정에 /sbin/nologin 쉘 부여 여부")
        details+=("취약")
        
        local formatted_accounts=$(IFS=, ; echo "${vulnerable_accounts[*]}")
        details+=("취약 계정 발견: [ $formatted_accounts ]. usermod -s /sbin/nologin <계정명> 명령으로 수정 필요")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-11" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then

    if [[ "$output_mode" == "all" ]]; then

# [U_17] 시스템 시작 스크립트 권한 설정 
# --------------------------------------------------------------------
U_17() {
    local desc="시스템 시작 스크립트 권한 설정"
    if [ ! -e /etc/rc.local ]; then # -f 대신 -e(존재 여부) 사용
        result_print "U-17" "$desc" "양호" "/etc/rc.local 파일 없음" "양호" "-"
    else
        # [중요] OWNER와 PERM 모두 -L 옵션을 사용하여 원본을 참조하도록 수정
        local OWNER=$(stat -L -c %U /etc/rc.local)
        local PERM=$(stat -L -c %a /etc/rc.local)
        
        local owner_res="양호"; local owner_req="-"
        local perm_res="양호"; local perm_req="-"

        if [ "$OWNER" != "root" ]; then
            owner_res="취약"; owner_req="chown root /etc/rc.d/rc.local"
        fi
        
        # 권한 비교 (PERM이 755보다 큰지 숫자 비교)
        if [ "$PERM" -gt 755 ]; then
            perm_res="취약"; perm_req="chmod 755 /etc/rc.d/rc.local"
        fi

        local final_res="양호"
        if [ "$owner_res" == "취약" ] || [ "$perm_res" == "취약" ]; then final_res="취약"; fi

        result_print "U-17" "$desc" "$final_res" \
        "소유자: $OWNER" "$owner_res" "$owner_req" \
        "권한: $PERM" "$perm_res" "$perm_req"
    fi
}

# [U_31] 홈 디렉토리 소유자 및 권한 설정
# --------------------------------------------------------------------
U_31() {
    local desc="홈 디렉터리 소유자 및 권한 설정"
    local BAD_HOME=0
    
 # 디렉토리가 없으면 '점검 대상 아님'으로 표시하게 변경
while IFS=: read -r user pass uid gid comment home shell; do
    if [ "$uid" -ge 1000 ]; then
        # nobody 계정 예외 처리
        if [ "$user" == "nobody" ]; then
            continue
        fi

        if [ ! -d "$home" ]; then
            continue
        elif [ "$(stat -c %U "$home")" != "$user" ]; then
            BAD_HOME=$((BAD_HOME+1))
        fi
    fi

# --------------------------------------------------------------------
# [U_66] 정책에 따른 시스템 로깅 설정
# --------------------------------------------------------------------
U_66()
{
    local output_mode=$1
    local desc="정책에 따른 시스템 로깅 설정"
    local total_result="양호"
    local details=()

    # 1. rsyslog(로그 기록기) 서비스가 살아서 돌고 있는지 확인
    local rsyslog_status=$(systemctl is-active rsyslog 2>/dev/null)

    if [[ "$rsyslog_status" != "active" ]]; then
        total_result="취약"
        details+=("rsyslog 서비스 활성화 여부")
        details+=("취약")
        details+=("rsyslog 서비스가 중지됨. 'systemctl start rsyslog' 명령으로 실행 필요")
    else
        details+=("rsyslog 서비스 활성화 여부")
        details+=("양호")
        details+=("-")
    fi

    # 2. 우분투 환경의 기본 로그 설정 파일들 경로
    local log_files="/etc/rsyslog.conf /etc/rsyslog.d/*.conf"
    
    # 3. 핵심 3대장(인증, 크론, 메일) 로그 기록이 주석(#) 처리 없이 켜져 있는지 검색
    local check_auth=$(grep -E "^\s*authpriv\.|^\s*auth\," $log_files 2>/dev/null)
    local check_cron=$(grep -E "^\s*cron\." $log_files 2>/dev/null)
    local check_mail=$(grep -E "^\s*mail\." $log_files 2>/dev/null)

    # 셋 중 하나라도 꺼져있다면 취약!
    if [[ -z "$check_auth" || -z "$check_cron" || -z "$check_mail" ]]; then
        total_result="취약"
        details+=("주요 로그(auth, cron, mail) 기록 설정 여부")
        details+=("취약")
        details+=("/etc/rsyslog.d/50-default.conf 등에서 authpriv, cron, mail 로그 기록 주석 해제 필요")
    else
        details+=("주요 로그(auth, cron, mail) 기록 설정 여부")
        details+=("양호")
        details+=("-")
    fi

    # 출력
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-66" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U-66" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
# [U_67] 로그 디렉터리 소유자 및 권한 설정
# --------------------------------------------------------------------
U_67()
{
    local output_mode=$1
    local desc="로그 디렉터리 소유자 및 권한 설정"
    local total_result="양호"
    local details=()

    # 우분투 및 리눅스 주요 로그 파일 리스트
    local log_files=("/var/log/syslog" "/var/log/auth.log" "/var/log/daemon.log" "/var/log/mail.log" "/var/log/messages" "/var/log/secure")
    local vulnerable_logs=()

    # 파일이 존재하는지 확인 후 소유자와 권한 체크
    for log in "${log_files[@]}"; do
        if [ -f "$log" ]; then
            # %U는 소유자 이름, %a는 644 같은 숫자 권한을 뽑아옵니다.
            local owner=$(stat -c "%U" "$log")
            local perm=$(stat -c "%a" "$log" | grep -o '...$') 
            
            # 소유자가 root 또는 syslog가 아니거나, 권한이 644보다 큰 경우 (취약)
            if [[ "$owner" != "root" && "$owner" != "syslog" ]] || [ "$perm" -gt 644 ]; then
                vulnerable_logs+=("$log(소유자:$owner/권한:$perm)")
            fi
        fi
    done

    # 취약한 파일이 하나도 없다면 양호!
    if [ ${#vulnerable_logs[@]} -eq 0 ]; then
        details+=("주요 로그 파일 소유자(root) 및 권한(644 이하) 설정")
        details+=("양호")
        details+=("-")
    else
        # 하나라도 걸리면 취약!
        total_result="취약"
        details+=("주요 로그 파일 소유자(root) 및 권한(644 이하) 설정")
        details+=("취약")
        
        local formatted_logs=$(IFS=, ; echo "${vulnerable_logs[*]}")
        details+=("취약 파일 발견: [ $formatted_logs ]. chown root 및 chmod 644 명령으로 조치 필요")
    fi

    # 출력 
    if [[ "$output_mode" == "all" ]]; then
        result_print "U-67" "$desc" "$total_result" "${details[@]}"
    elif [[ "$output_mode" == "vulnerable" && "$total_result" == "취약" ]]; then
        result_print "U-67" "$desc" "$total_result" "${details[@]}"
    fi
}

# --------------------------------------------------------------------
main() {
    # 1. 인자값(all 또는 vulnerable)을 받음. 없으면 기본값은 'all'
    local mode=${1:-"all"}

    # 1. 1번부터 67번까지 순차 실행
    for i in {1..67}; do
        local func_name=$(printf "U_%02d" $i)
        if declare -f "$func_name" > /dev/null; then
            # 'all' 대신 변수 $mode를 사용
            "$func_name" "$mode"
        fi
    done

    # 2. 요약 정보 출력 (사진의 디자인 반영)
echo -e "\e[34m==========================================================\e[0m"
echo -e "\e[34m==========================================================\e[0m"

echo -e "\e[34m==========================================================\e[0m"
echo -e "\e[34m==========================================================\e[0m\n"
}
